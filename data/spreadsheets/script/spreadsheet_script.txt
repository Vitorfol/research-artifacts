// Função genérica para ordenar a tabela
function ordenarTabela(colunaIndex) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();

  // Definir o intervalo das linhas a partir da linha 45 até a última linha
  const range = sheet.getRange(2, 1, sheet.getLastRow() - 1, sheet.getLastColumn());

  // Ordenar pelo número da coluna passada como parâmetro
  range.sort({ column: colunaIndex, ascending: true });
}
